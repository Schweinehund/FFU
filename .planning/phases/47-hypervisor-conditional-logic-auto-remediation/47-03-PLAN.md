Add a new section in BuildFFUVM_UI.ps1 AFTER the dashboard poll timer code block (after the `$script:uiState.Data.dashboardPollTimer.Start()` line) and BEFORE the refresh button handler. This section wires click handlers for dynamically created Fix and Copy buttons.

**The approach:** Since Fix/Copy buttons are dynamically created by `Update-DashboardCheckUI` during poll timer processing, we cannot wire handlers at creation time from the module. Instead, we handle this with a helper function that the poll timer calls after each `Update-DashboardCheckUI` call to wire newly created buttons.

**A. Create Wire-DashboardButtonHandlers function:**

```powershell
function Wire-DashboardButtonHandlers {
    <#
    .SYNOPSIS
        Wires click handlers for dynamically created Fix and Copy buttons in the dashboard.
    .DESCRIPTION
        Scans dashboard category panels for buttons named btnFix_*, btnFixUnsafe_*, and
        btnCopy_* that don't yet have handlers, and attaches appropriate click event handlers.
    #>
    param(
        [Parameter(Mandatory)]
        [PSCustomObject]$State
    )

    foreach ($category in @('System', 'Hypervisor', 'BuildTools', 'Network', 'Optimization')) {
        $panel = $State.Controls."pnl${category}Checks"
        if ($null -eq $panel) { continue }

        # Recursively find all buttons in the panel
        $buttons = Find-WPFButtons -Element $panel

        foreach ($btn in $buttons) {
            # Skip if already wired (use Tag to track)
            if ($btn.Tag -is [hashtable] -and $btn.Tag.Wired) { continue }

            $btnName = $btn.Name
            if ([string]::IsNullOrWhiteSpace($btnName)) { continue }

            if ($btnName -like 'btnFix_*') {
                # Safe fix button
                $checkName = $btnName.Substring('btnFix_'.Length)
                Wire-SafeFixHandler -Button $btn -CheckName $checkName -State $State
            }
            elseif ($btnName -like 'btnFixUnsafe_*') {
                # Unsafe fix button
                $checkName = $btnName.Substring('btnFixUnsafe_'.Length)
                Wire-UnsafeFixHandler -Button $btn -CheckName $checkName -State $State
            }
            elseif ($btnName -like 'btnCopy_*') {
                # Copy button
                $checkName = $btnName.Substring('btnCopy_'.Length)
                Wire-CopyHandler -Button $btn -CheckName $checkName -State $State
            }
        }
    }
}
```

**B. Create Find-WPFButtons helper** (recursive panel walker):
```powershell
function Find-WPFButtons {
    param([System.Windows.UIElement]$Element)
    $buttons = @()
    if ($Element -is [System.Windows.Controls.Button]) {
        $buttons += $Element
    }
    if ($Element -is [System.Windows.Controls.Panel]) {
        foreach ($child in $Element.Children) {
            $buttons += Find-WPFButtons -Element $child
        }
    }
    if ($Element -is [System.Windows.Controls.ContentControl] -and $null -ne $Element.Content) {
        if ($Element.Content -is [System.Windows.UIElement]) {
            $buttons += Find-WPFButtons -Element $Element.Content
        }
    }
    return $buttons
}
```

**C. Create Wire-SafeFixHandler** — handles safe one-click fix (REM-01):
```powershell
function Wire-SafeFixHandler {
    param(
        [System.Windows.Controls.Button]$Button,
        [string]$CheckName,
        [PSCustomObject]$State
    )
    # Mark as wired
    $Button.Tag = @{ Wired = $true; CheckName = $CheckName }

    $Button.Add_Click({
        param($sender, $e)
        $cn = $sender.Tag.CheckName

        # Disable button and show fixing state
        $sender.Content = 'Fixing...'
        $sender.IsEnabled = $false
        $sender.Background = [System.Windows.Media.BrushConverter]::new().ConvertFromString('#E0E0E0')

        # Execute repair in ThreadJob to avoid blocking UI
        $repairJob = Start-ThreadJob -ScriptBlock {
            param($checkName, $ffuPath)
            $modulePath = Join-Path $ffuPath 'Modules'
            if ($env:PSModulePath -notlike "*$modulePath*") {
                $env:PSModulePath = "$modulePath;$env:PSModulePath"
            }
            $ffuuiPath = Join-Path $ffuPath 'FFUUI.Core'
            Import-Module (Join-Path $ffuuiPath 'FFUUI.Core.psd1') -Force -ErrorAction SilentlyContinue
            Import-Module FFU.Preflight -Force

            Invoke-DashboardRemediation -CheckName $checkName -FFUDevelopmentPath $ffuPath
        } -ArgumentList $cn, $script:FFUDevelopmentPath

        # Poll for repair completion using DispatcherTimer (500ms)
        $repairTimer = [System.Windows.Threading.DispatcherTimer]::new()
        $repairTimer.Interval = [TimeSpan]::FromMilliseconds(500)
        $repairTimer.Tag = @{ Job = $repairJob; Button = $sender; CheckName = $cn }
        $repairTimer.Add_Tick({
            param($timerSender, $timerE)
            $job = $timerSender.Tag.Job
            $btn = $timerSender.Tag.Button
            $cn = $timerSender.Tag.CheckName

            if ($job.State -in @('Completed', 'Failed', 'Stopped')) {
                $timerSender.Stop()
                $repairResult = $null
                try {
                    $repairResult = Receive-Job -Job $job -ErrorAction SilentlyContinue
                    Remove-Job -Job $job -Force -ErrorAction SilentlyContinue
                } catch {}

                $succeeded = $false
                if ($null -ne $repairResult -and $repairResult.PSObject.Properties['Succeeded']) {
                    $succeeded = $repairResult.Succeeded
                }

                if ($succeeded) {
                    # Re-run just this single check to update status
                    # Re-enable button with success state temporarily
                    $btn.Content = 'Fixed!'
                    $btn.Background = [System.Windows.Media.BrushConverter]::new().ConvertFromString('#C8E6C9')

                    # Re-run the single check in background and update UI
                    Invoke-SingleCheckRefresh -CheckName $cn -State $script:uiState
                } else {
                    # Repair failed — re-enable Fix button for retry
                    $btn.Content = 'Fix'
                    $btn.IsEnabled = $true
                    $btn.Background = [System.Windows.Media.BrushConverter]::new().ConvertFromString('#C8E6C9')
                }
            }
        })
        $repairTimer.Start()
    }.GetNewClosure())
}
```

**D. Create Wire-UnsafeFixHandler** — handles reboot-required fix (REM-02):

```powershell
function Wire-UnsafeFixHandler {
    param(
        [System.Windows.Controls.Button]$Button,
        [string]$CheckName,
        [PSCustomObject]$State
    )
    $Button.Tag = @{ Wired = $true; CheckName = $CheckName }

    $Button.Add_Click({
        param($sender, $e)
        $cn = $sender.Tag.CheckName
        $unsafeMap = Get-UnsafeRemediationMap
        $info = $unsafeMap[$cn]
        if ($null -eq $info) { return }

        $result = [System.Windows.MessageBox]::Show(
            $info.ConfirmMessage,
            $info.ConfirmTitle,
            [System.Windows.MessageBoxButton]::YesNo,
            [System.Windows.MessageBoxImage]::Warning
        )

        if ($result -eq [System.Windows.MessageBoxResult]::Yes) {
            $sender.Content = 'Fixing...'
            $sender.IsEnabled = $false

            # Execute the command
            try {
                Invoke-Expression $info.Command
                [System.Windows.MessageBox]::Show(
                    $info.SuccessMessage,
                    'Reboot Required',
                    [System.Windows.MessageBoxButton]::OK,
                    [System.Windows.MessageBoxImage]::Information
                )
            } catch {
                [System.Windows.MessageBox]::Show(
                    "Failed to execute: $($_.Exception.Message)",
                    'Error',
                    [System.Windows.MessageBoxButton]::OK,
                    [System.Windows.MessageBoxImage]::Error
                )
                $sender.Content = 'Fix...'
                $sender.IsEnabled = $true
            }
        } else {
            # User declined — show "Fix available" indicator
            if ($null -ne $script:uiState.Data) {
                $script:uiState.Data."fixDeclined_$cn" = $true
            }
            # Change button to "Fix available" text indicator
            $sender.Content = 'Fix available'
            $sender.FontStyle = [System.Windows.FontStyles]::Italic
            $sender.Foreground = [System.Windows.Media.BrushConverter]::new().ConvertFromString('#1565C0')
            $sender.Background = [System.Windows.Media.Brushes]::Transparent
            $sender.BorderThickness = [System.Windows.Thickness]::new(0)
        }
    }.GetNewClosure())
}
```

**E. Create Wire-CopyHandler** — handles clipboard copy (REM-03):

```powershell
function Wire-CopyHandler {
    param(
        [System.Windows.Controls.Button]$Button,
        [string]$CheckName,
        [PSCustomObject]$State
    )
    $Button.Tag = @{ Wired = $true; CheckName = $CheckName }

    $Button.Add_Click({
        param($sender, $e)
        $cn = $sender.Tag.CheckName

        # Find the associated TextBox
        $parent = $sender.Parent
        if ($null -ne $parent) {
            foreach ($child in $parent.Children) {
                if ($child -is [System.Windows.Controls.TextBox] -and $child.Name -eq "txtRemediation_$cn") {
                    try {
                        [System.Windows.Clipboard]::SetText($child.Text)
                        $originalContent = $sender.Content
                        $sender.Content = 'Copied!'

                        # Reset after 2 seconds using DispatcherTimer (async, per user decision)
                        $resetTimer = [System.Windows.Threading.DispatcherTimer]::new()
                        $resetTimer.Interval = [TimeSpan]::FromSeconds(2)
                        $resetTimer.Tag = @{ Button = $sender; OriginalContent = $originalContent }
                        $resetTimer.Add_Tick({
                            param($ts, $te)
                            $ts.Tag.Button.Content = $ts.Tag.OriginalContent
                            $ts.Stop()
                        })
                        $resetTimer.Start()
                    }
                    catch {
                        $sender.Content = 'Error!'
                        $sender.ToolTip = "Copy failed: $($_.Exception.Message)"
                    }
                    break
                }
            }
        }
    }.GetNewClosure())
}
```

```powershell
function Invoke-SingleCheckRefresh {
    param(
        [Parameter(Mandatory)]
        [string]$CheckName,
        [PSCustomObject]$State
    )
    # Launch a ThreadJob that runs just the single Test-FFU* function
    $refreshJob = Start-ThreadJob -ScriptBlock {
        param($checkName, $ffuPath)
        $modulePath = Join-Path $ffuPath 'Modules'
        if ($env:PSModulePath -notlike "*$modulePath*") {
            $env:PSModulePath = "$modulePath;$env:PSModulePath"
        }
        Import-Module FFU.Preflight -Force

        $funcName = "Test-FFU$checkName"
        $result = & $funcName
        return $result
    } -ArgumentList $CheckName, $script:FFUDevelopmentPath

    # Poll for completion
    $refreshTimer = [System.Windows.Threading.DispatcherTimer]::new()
    $refreshTimer.Interval = [TimeSpan]::FromMilliseconds(500)
    $refreshTimer.Tag = @{ Job = $refreshJob; CheckName = $CheckName; State = $State }
    $refreshTimer.Add_Tick({
        param($ts, $te)
        $job = $ts.Tag.Job
        $cn = $ts.Tag.CheckName
        $state = $ts.Tag.State

        if ($job.State -in @('Completed', 'Failed', 'Stopped')) {
            $ts.Stop()
            $checkResult = $null
            try {
                $checkResult = Receive-Job -Job $job -ErrorAction SilentlyContinue
                Remove-Job -Job $job -Force -ErrorAction SilentlyContinue
            } catch {}

            if ($null -ne $checkResult) {
                # Clear previous check display from category panel and re-render
                $category = Get-CheckCategory -CheckName $cn
                $panel = $state.Controls."pnl${category}Checks"
                if ($null -ne $panel) {
                    # Find and remove the existing check entry (look for outerPanel containing this check)
                    # Simpler approach: clear panel and re-add all checks from stats
                    # For single check refresh, just add the updated result below existing ones
                    # The user will see the updated status — full re-render on next Refresh click

                    $severity = if ($checkResult.PSObject.Properties['Severity']) { $checkResult.Severity } else { 'Info' }
                    $remediation = if ($checkResult.PSObject.Properties['Remediation']) { $checkResult.Remediation } else { '' }
                    $durationMs = if ($checkResult.PSObject.Properties['DurationMs']) { $checkResult.DurationMs } else { 0 }

                    Update-DashboardCheckUI -State $state -CheckName $cn `
                        -Status $checkResult.Status -Severity $severity `
                        -Message $checkResult.Message -Remediation $remediation `
                        -DurationMs $durationMs

                    # Wire handlers for newly created buttons
                    Wire-DashboardButtonHandlers -State $state
                }
            }
        }
    })
    $refreshTimer.Start()
}
```
