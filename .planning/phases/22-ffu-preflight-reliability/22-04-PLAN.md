    ```powershell
    function New-FFUCheckResult {
        [CmdletBinding()]
        [OutputType([PSCustomObject])]
        param(
            [Parameter(Mandatory = $true)]
            [string]$CheckName,

            [Parameter(Mandatory = $true)]
            [ValidateSet('Passed', 'Failed', 'Warning', 'Skipped')]
            [string]$Status,

            [Parameter(Mandatory = $true)]
            [string]$Message,

            [Parameter()]
            [hashtable]$Details = @{},

            [Parameter()]
            [string]$Remediation = '',

            [Parameter()]
            [int]$DurationMs = 0,

            # NEW: Severity classification
            [Parameter()]
            [ValidateSet('Critical', 'Warning', 'Info')]
            [string]$Severity = 'Warning'
        )

        [PSCustomObject]@{
            CheckName   = $CheckName
            Status      = $Status
            Severity    = $Severity  # NEW
            Message     = $Message
            Details     = $Details
            Remediation = $Remediation
            DurationMs  = $DurationMs
        }
    }
    ```

    - **Info**: Informational only, no action needed (vmxtoolkit optional, DISM cleanup)

    **Add helper function for severity tracking (internal):**
    ```powershell
    function Add-CheckToResult {
        param(
            [PSCustomObject]$Result,
            [PSCustomObject]$CheckResult,
            [string]$TierKey,
            [string]$ResultKey
        )

        # Add to tier results
        $Result."Tier${TierKey}Results"[$ResultKey] = $CheckResult

        # Track severity counts for failures
        if ($CheckResult.Status -eq 'Failed') {
            switch ($CheckResult.Severity) {
                'Critical' {
                    $Result.CriticalCount++
                    $Result.Errors.Add("[$($CheckResult.Severity.ToUpper())] $ResultKey: $($CheckResult.Message)")
                }
                'Warning' {
                    $Result.WarningCount++
                    $Result.Warnings.Add("[$($CheckResult.Severity.ToUpper())] $ResultKey: $($CheckResult.Message)")
                }
                'Info' {
                    $Result.InfoCount++
                    $Result.InfoMessages.Add("[$($CheckResult.Severity.ToUpper())] $ResultKey: $($CheckResult.Message)")
                }
            }

            # Only block on Critical failures
            if ($CheckResult.Severity -eq 'Critical') {
                $Result.IsValid = $false
            }

            # Add remediation if present
            if ($CheckResult.Remediation) {
                $Result.RemediationSteps.Add($CheckResult.Remediation)
            }
        }
        elseif ($CheckResult.Status -eq 'Warning') {
            $Result.HasWarnings = $true
            switch ($CheckResult.Severity) {
                'Critical' { $Result.CriticalCount++ }
                'Warning' { $Result.WarningCount++ }
                'Info' { $Result.InfoCount++ }
            }
            $Result.Warnings.Add("[$($CheckResult.Severity.ToUpper())] $ResultKey: $($CheckResult.Message)")
        }
    }
    ```

    **Update summary output at end of Invoke-FFUPreflight:**
    ```powershell
    #region Summary Output
    $overallStopwatch.Stop()
    $result.ValidationDurationMs = $overallStopwatch.ElapsedMilliseconds

    Write-Information "`n========================================"
    Write-Information "   Pre-Flight Validation Summary"
    Write-Information "========================================`n"

    # Show severity breakdown
    if ($result.CriticalCount -gt 0) {
        Write-Information "CRITICAL ISSUES: $($result.CriticalCount) (build cannot proceed)"
        foreach ($error in $result.Errors | Where-Object { $_ -match '^\[CRITICAL\]' }) {
            Write-Information "  - $($error -replace '^\[CRITICAL\] ', '')"
        }
        Write-Information ""
    }

    if ($result.WarningCount -gt 0) {
        Write-Information "WARNINGS: $($result.WarningCount) (build may have issues)"
        foreach ($warning in $result.Warnings | Where-Object { $_ -match '^\[WARNING\]' }) {
            Write-Information "  - $($warning -replace '^\[WARNING\] ', '')"
        }
        Write-Information ""
    }

    if ($result.InfoCount -gt 0) {
        Write-Information "INFO: $($result.InfoCount) (optional improvements)"
        foreach ($info in $result.InfoMessages) {
            Write-Information "  - $($info -replace '^\[INFO\] ', '')"
        }
        Write-Information ""
    }

    # Overall status
    if ($result.IsValid) {
        if ($result.HasWarnings) {
            Write-Information "RESULT: PASSED with warnings"
            Write-Information "The build can proceed, but review warnings above."
        }
        else {
            Write-Information "RESULT: PASSED"
            Write-Information "All pre-flight checks passed. Ready to build."
        }
    }
    else {
        Write-Information "RESULT: FAILED"
        Write-Information "Fix critical issues above before proceeding."
        Write-Information ""
        Write-Information "Run with -Verbose for detailed remediation steps."
    }

    Write-Information "`nValidation completed in $($result.ValidationDurationMs)ms"
    Write-Information "========================================"
    #endregion
    ```
