**Describe 'Update-BuildButtonState' (mock-based):**

Since WPF controls can't be created in Pester without a dispatcher, test the logic by mocking $State:
```powershell
It 'should set dashboardWarningCount in state data for warnings' {
    # Create mock state with minimal structure
    $mockState = @{
        Controls = @{
            btnRun = [PSCustomObject]@{
                IsEnabled = $true
                ToolTip = $null
            }
        }
        Data = @{}
    }

    Update-BuildButtonState -State $mockState -CriticalCount 0 -WarningCount 3

    $mockState.Data.dashboardWarningCount | Should -Be 3
}

It 'should disable button for critical failures' {
    $mockState = @{
        Controls = @{
            btnRun = [PSCustomObject]@{
                IsEnabled = $true
                ToolTip = $null
            }
        }
        Data = @{}
    }

    Update-BuildButtonState -State $mockState -CriticalCount 2 -WarningCount 1

    $mockState.Controls.btnRun.IsEnabled | Should -BeFalse
}

It 'should enable button when all checks pass' {
    $mockState = @{
        Controls = @{
            btnRun = [PSCustomObject]@{
                IsEnabled = $false
                ToolTip = 'previous tooltip'
            }
        }
        Data = @{ dashboardWarningCount = 5 }
    }

    Update-BuildButtonState -State $mockState -CriticalCount 0 -WarningCount 0

    $mockState.Controls.btnRun.IsEnabled | Should -BeTrue
    $mockState.Data.dashboardWarningCount | Should -Be 0
}
```
