$copyFlags = @{

    CopyPPKG      = $true

    CopyAutopilot = $true
    CopyAppsISO   = $true

if ($null -ne $config -and $null -ne $config.USBMode -and $null -ne $config.USBMode.Artifacts) {
    $configArtifacts = $config.USBMode.Artifacts

    # Map artifact types to their config entries and manifest properties
    $artifactTypeMap = @{
        FFU       = @{ manifestProp = 'FFUFiles';       copyFlag = 'CopyFFU' }
        DeployISO = @{ manifestProp = 'DeployISO';      copyFlag = $null }  # Always required
        Drivers   = @{ manifestProp = 'Drivers';        copyFlag = 'CopyDrivers' }
        PPKG      = @{ manifestProp = 'PPKGFiles';      copyFlag = 'CopyPPKG' }
        Unattend  = @{ manifestProp = 'UnattendFiles';  copyFlag = 'CopyUnattend' }
        Autopilot = @{ manifestProp = 'AutopilotFiles'; copyFlag = 'CopyAutopilot' }
        AppsISO   = @{ manifestProp = 'AppsISO';        copyFlag = 'CopyAppsISO' }
    }

    foreach ($type in $artifactTypeMap.Keys) {
        $entry = $configArtifacts.$type
        if ($null -eq $entry) { continue }

        # Path override (D-30): user-browsed path replaces auto-detected
        if (-not [string]::IsNullOrWhiteSpace($entry.Path) -and (Test-Path -LiteralPath $entry.Path)) {
            WriteLog "USBOnlyMode: Using user-specified $type path: $($entry.Path)"
            # Update the manifest or set path variable directly.

        # Include flag (D-29): checkbox state controls whether this artifact is copied
        $flagName = $artifactTypeMap[$type].copyFlag
        if ($null -ne $flagName -and $entry.PSObject.Properties.Match('Include').Count -gt 0) {

            if (-not $copyFlags[$flagName]) {
