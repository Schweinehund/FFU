```powershell
function Test-CheckpointArtifacts {
    <#
    .SYNOPSIS
    Validates that checkpoint artifact paths still exist on disk.

    .DESCRIPTION
    Before resuming from a checkpoint, this function verifies that the critical
    artifact paths recorded in the checkpoint (VHDX, FFU, ISO files) still exist.
    If artifacts are missing, the checkpoint is considered invalid and a fresh
    build should be started.

    This prevents resume failures when:
    - User manually deleted files between runs
    - Cleanup scripts ran after termination
    - Files were moved to different location

    .PARAMETER Checkpoint
    The checkpoint object returned by Get-FFUBuildCheckpoint.

    .OUTPUTS
    PSCustomObject with:
    - Valid (bool): Whether all critical artifacts exist
    - MissingPaths (string[]): List of paths that don't exist
    - CheckedPaths (string[]): List of all paths that were checked

    .EXAMPLE
    $checkpoint = Get-FFUBuildCheckpoint -FFUDevelopmentPath 'C:\FFUDev'
    $validation = Test-CheckpointArtifacts -Checkpoint $checkpoint
    if (-not $validation.Valid) {
        Write-Warning "Missing artifacts: $($validation.MissingPaths -join ', ')"
    }
    #>
    [CmdletBinding()]
    [OutputType([PSCustomObject])]
    param(
        [Parameter(Mandatory)]
        [PSCustomObject]$Checkpoint
    )

    $result = [PSCustomObject]@{
        Valid        = $true
        MissingPaths = @()
        CheckedPaths = @()
    }

    # Define which paths are critical based on completed phase
    $pathsToCheck = @()

    # Check paths hashtable from checkpoint
    if ($null -ne $Checkpoint.paths) {
        $paths = $Checkpoint.paths

        # Always check FFUDevelopmentPath
        if ($paths.FFUDevelopmentPath) {
            $pathsToCheck += $paths.FFUDevelopmentPath
        }

        # Check VHDX if we're past VHDX creation
        $vhdxPhases = @('VHDXCreation', 'WindowsUpdates', 'VMSetup', 'VMStart', 'AppInstallation', 'VMShutdown', 'FFUCapture')
        if ($Checkpoint.lastCompletedPhase -in $vhdxPhases -and $paths.VHDXPath) {
            $pathsToCheck += $paths.VHDXPath
        }

        # Check FFU if we're past FFU capture
        $ffuPhases = @('FFUCapture', 'DeploymentMedia', 'USBCreation')
        if ($Checkpoint.lastCompletedPhase -in $ffuPhases) {
            if ($paths.FFUFile) { $pathsToCheck += $paths.FFUFile }
            if ($paths.FFUCaptureLocation) { $pathsToCheck += $paths.FFUCaptureLocation }
        }

        # Check deployment media if created
        if ($Checkpoint.lastCompletedPhase -in @('DeploymentMedia', 'USBCreation')) {
            if ($paths.DeployISO -and $Checkpoint.artifacts.deploymentMediaCreated) {
                $pathsToCheck += $paths.DeployISO
            }
        }
    }

    # Validate each path
    foreach ($path in $pathsToCheck) {
        if (-not [string]::IsNullOrWhiteSpace($path)) {
            $result.CheckedPaths += $path
            if (-not (Test-Path -LiteralPath $path)) {
                $result.Valid = $false
                $result.MissingPaths += $path
            }
        }
    }

    return $result
}
```

```powershell
#Requires -Modules @{ ModuleName = 'Pester'; ModuleVersion = '5.0.0' }

Describe 'FFU.Checkpoint Resume Validation' -Tag 'Unit', 'FFU.Checkpoint', 'REL-BUILD-03' {
    BeforeAll {
        $modulePath = Join-Path $PSScriptRoot '..\..\FFUDevelopment\Modules\FFU.Checkpoint'
        Import-Module $modulePath -Force

        # Create temp directory for test artifacts
        $script:TestDir = Join-Path $env:TEMP "FFUCheckpointTests_$(Get-Random)"
        New-Item -Path $script:TestDir -ItemType Directory -Force | Out-Null
    }

    AfterAll {
        # Cleanup
        if (Test-Path $script:TestDir) {
            Remove-Item -Path $script:TestDir -Recurse -Force -ErrorAction SilentlyContinue
        }
    }

    Context 'Test-CheckpointArtifacts with valid paths' {
        It 'Should return Valid=true when FFUDevelopmentPath exists' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'PreflightValidation'
                paths = @{ FFUDevelopmentPath = $script:TestDir }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.Valid | Should -Be $true
        }

        It 'Should check VHDX path for VHDXCreation phase' {
            $vhdxPath = Join-Path $script:TestDir 'test.vhdx'
            New-Item -Path $vhdxPath -ItemType File -Force | Out-Null

            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'VHDXCreation'
                paths = @{
                    FFUDevelopmentPath = $script:TestDir
                    VHDXPath = $vhdxPath
                }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.Valid | Should -Be $true
            $result.CheckedPaths | Should -Contain $vhdxPath
        }
    }

    Context 'Test-CheckpointArtifacts with missing paths' {
        It 'Should return Valid=false when FFUDevelopmentPath missing' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'PreflightValidation'
                paths = @{ FFUDevelopmentPath = 'C:\NonExistent\Path\12345' }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.Valid | Should -Be $false
            $result.MissingPaths | Should -Contain 'C:\NonExistent\Path\12345'
        }

        It 'Should return Valid=false when VHDX path missing' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'VHDXCreation'
                paths = @{
                    FFUDevelopmentPath = $script:TestDir
                    VHDXPath = 'C:\NonExistent\test.vhdx'
                }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.Valid | Should -Be $false
        }

        It 'Should list all missing paths' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'FFUCapture'
                paths = @{
                    FFUDevelopmentPath = 'C:\Missing1'
                    FFUCaptureLocation = 'C:\Missing2'
                }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.MissingPaths.Count | Should -BeGreaterThan 0
        }
    }

    Context 'Test-CheckpointArtifacts phase-aware validation' {
        It 'Should not check VHDX for PreflightValidation phase' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'PreflightValidation'
                paths = @{
                    FFUDevelopmentPath = $script:TestDir
                    VHDXPath = 'C:\NonExistent\test.vhdx'  # Missing but shouldn't matter
                }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            # Should be valid because VHDX isn't checked for early phases
            $result.CheckedPaths | Should -Not -Contain 'C:\NonExistent\test.vhdx'
        }

        It 'Should check FFU for FFUCapture phase' {
            $ffuDir = Join-Path $script:TestDir 'FFU'
            New-Item -Path $ffuDir -ItemType Directory -Force | Out-Null

            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'FFUCapture'
                paths = @{
                    FFUDevelopmentPath = $script:TestDir
                    FFUCaptureLocation = $ffuDir
                }
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.CheckedPaths | Should -Contain $ffuDir
        }
    }

    Context 'Edge cases' {
        It 'Should handle null paths gracefully' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'PreflightValidation'
                paths = @{ FFUDevelopmentPath = $null }
                artifacts = @{}
            }
            { Test-CheckpointArtifacts -Checkpoint $checkpoint } | Should -Not -Throw
        }

        It 'Should handle empty paths hashtable' {
            $checkpoint = [PSCustomObject]@{
                lastCompletedPhase = 'PreflightValidation'
                paths = @{}
                artifacts = @{}
            }
            $result = Test-CheckpointArtifacts -Checkpoint $checkpoint
            $result.Valid | Should -Be $true
            $result.CheckedPaths.Count | Should -Be 0
        }
    }
}
```
