    **Function signature:**
    ```powershell
    function Test-MSUIntegrity {
        [CmdletBinding()]
        [OutputType([PSCustomObject])]
        param(
            [Parameter(Mandatory)]
            [string]$FilePath,

            [Parameter()]
            [string]$ExpectedHash,

            [Parameter()]
            [int64]$ExpectedSize,

            [Parameter()]
            [int64]$MinimumSizeBytes = 1MB
        )
    }
    ```

    **Implementation:**
    1. Initialize result object: `@{ Valid = $true; FilePath = $FilePath; Errors = [System.Collections.Generic.List[string]]::new() }`
    2. Check file exists - if not, add error and return early with Valid = $false
    3. Get file info: `$fileInfo = Get-Item $FilePath`
    4. Check for empty file (0 bytes) - corruption indicator, add error
    5. Check minimum size - if < MinimumSizeBytes, add error "File is suspiciously small"
    6. Check expected size if provided - if mismatch, add error with both values
    7. Check hash if ExpectedHash provided:
       - Use pattern from Get-ProductsCab: `[System.Security.Cryptography.SHA256]::Create()`
       - Open file stream, compute hash, dispose properly in finally block
       - Convert to Base64: `[Convert]::ToBase64String($hashBytes)`
       - If mismatch, add error with both values
    8. Set Valid = $false if any errors added
    9. Return `[PSCustomObject]$result`

    **Add to FunctionsToExport in .psd1:**

    **Add validation before returning filename:**
    ```powershell
    # After successful download, validate integrity
    $downloadedPath = Join-Path -Path $Path -ChildPath $fileName
    $integrityCheck = Test-MSUIntegrity -FilePath $downloadedPath -MinimumSizeBytes 1MB

    if (-not $integrityCheck.Valid) {
        WriteLog "WARNING: Downloaded file failed integrity check: $($integrityCheck.Errors -join '; ')"

        # Attempt re-download (one retry)
        WriteLog "Attempting re-download due to corruption..."
        Remove-Item -Path $downloadedPath -Force -ErrorAction SilentlyContinue

        try {
            Start-BitsTransferWithRetry -Source $link -Destination $Path -ErrorAction Stop | Out-Null
            $retryCheck = Test-MSUIntegrity -FilePath $downloadedPath -MinimumSizeBytes 1MB

            if (-not $retryCheck.Valid) {
                WriteLog "ERROR: Re-download also failed integrity check: $($retryCheck.Errors -join '; ')"
                WriteLog "RESOLUTION: The download source may be corrupted. Try:"
                WriteLog "  1. Clear browser/system download cache"
                WriteLog "  2. Try a different network connection"
                WriteLog "  3. Manually download the update from catalog.update.microsoft.com"
                Remove-Item -Path $downloadedPath -Force -ErrorAction SilentlyContinue
                continue  # Try next link
            }
            WriteLog "Re-download succeeded, integrity check passed"
        }
        catch {
            WriteLog "ERROR: Re-download failed: $($_.Exception.Message)"
            continue  # Try next link
        }
    }

    WriteLog "Download complete and validated: $fileName"
    return $fileName
    ```

    ```powershell
    Describe 'REL-UPD-02: MSU Download Validation' -Tag 'Unit', 'FFU.Updates', 'Reliability' {

        Context 'Test-MSUIntegrity function' {

            BeforeAll {
                # Create test files in TestDrive
                $validFile = Join-Path $TestDrive 'valid.msu'
                [byte[]]$content = 1..2MB | ForEach-Object { [byte](Get-Random -Maximum 256) }
                [System.IO.File]::WriteAllBytes($validFile, $content)

                $emptyFile = Join-Path $TestDrive 'empty.msu'
                [System.IO.File]::WriteAllBytes($emptyFile, [byte[]]@())

                $smallFile = Join-Path $TestDrive 'small.msu'
                [System.IO.File]::WriteAllBytes($smallFile, [byte[]](1..100))
            }

            It 'Should return Valid=true for file meeting minimum size' {
                $result = Test-MSUIntegrity -FilePath $validFile
                $result.Valid | Should -Be $true
                $result.Errors.Count | Should -Be 0
            }

            It 'Should return Valid=false for non-existent file' {
                $result = Test-MSUIntegrity -FilePath 'C:\nonexistent\file.msu'
                $result.Valid | Should -Be $false
                $result.Errors | Should -Contain 'File not found*'
            }

            It 'Should return Valid=false for empty file (0 bytes)' {
                $result = Test-MSUIntegrity -FilePath $emptyFile
                $result.Valid | Should -Be $false
                $result.Errors | Should -Match 'empty.*0 bytes'
            }

            It 'Should return Valid=false for file below minimum size' {
                $result = Test-MSUIntegrity -FilePath $smallFile -MinimumSizeBytes 1KB
                $result.Valid | Should -Be $false
                $result.Errors | Should -Match 'suspiciously small'
            }

            It 'Should validate expected size when provided' {
                $actualSize = (Get-Item $validFile).Length
                $result = Test-MSUIntegrity -FilePath $validFile -ExpectedSize ($actualSize + 1000)
                $result.Valid | Should -Be $false
                $result.Errors | Should -Match 'Size mismatch'
            }

            It 'Should validate hash when ExpectedHash provided' {
                # Compute correct hash
                $sha256 = [System.Security.Cryptography.SHA256]::Create()
                $fs = [System.IO.File]::OpenRead($validFile)
                $correctHash = [Convert]::ToBase64String($sha256.ComputeHash($fs))
                $fs.Dispose()

                # Test with correct hash
                $result = Test-MSUIntegrity -FilePath $validFile -ExpectedHash $correctHash
                $result.Valid | Should -Be $true

                # Test with wrong hash
                $result = Test-MSUIntegrity -FilePath $validFile -ExpectedHash 'wronghash'
                $result.Valid | Should -Be $false
                $result.Errors | Should -Match 'Hash mismatch'
            }
        }

        Context 'Save-KB integration' {

            It 'Should have integrity validation in implementation' {
                $source = (Get-Command Save-KB).ScriptBlock.ToString()
                $source | Should -Match 'Test-MSUIntegrity'
            }
        }
    }
    ```
