3. **Range validation:**
```powershell

        'integer' { return $SchemaProp.minimum ?? 1 }
```

```powershell
Context 'Actionable Error Messages' -Tag 'Actionable' {
    It 'Invalid enum value includes valid options and fix guidance' {

        $result = Test-FFUConfiguration -ConfigObject $config

        $result.IsValid | Should -BeFalse
        $result.Errors[0] | Should -Match 'Valid values:'

    }

    It 'Type mismatch includes example of correct format' {
        $config = @{ Memory = 'eight gigabytes' }
        $result = Test-FFUConfiguration -ConfigObject $config

        $result.IsValid | Should -BeFalse
        $result.Errors[0] | Should -Match 'expected.*integer'
        $result.Errors[0] | Should -Match 'To fix:'
    }

    It 'Unknown property suggests similar property name' {
        $config = @{ windowsSKU = 'Pro' }  # lowercase w
        $result = Test-FFUConfiguration -ConfigObject $config

        # Should suggest WindowsSKU

    It 'Pattern violation shows expected format' {
        $config = @{ ShareName = 'not-a-unc-path' }
        $result = Test-FFUConfiguration -ConfigObject $config

        $result.IsValid | Should -BeFalse
        $result.Errors[0] | Should -Match 'Expected format:'
    }
}
```
